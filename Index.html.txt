<script>
function go(id){
 document.querySelectorAll('.screen').forEach(s=>s.classList.remove('active'));
 document.getElementById(id).classList.add('active');
}

function saveClient(){
 localStorage.setItem('client', document.getElementById('clientName').value);
 go('services');
}

function selectService(s){
 localStorage.setItem('service', s);
 go('packages');
}

function placeOrder(price){
 let fee = price * 0.055;
 document.getElementById('priceInfo').innerText =
  'Total payable: R' + Math.round(price + fee);
 go('checkout');
}

function pay(){
 localStorage.setItem('orderStatus', 'escrow');
 go('clientDashboard');
}

function saveCleaner(){
 document.getElementById('jobAlert').innerText =
  'New job request: ' + localStorage.getItem('service');
 go('cleanerDash');
}

function deliver(){
 alert('Work delivered to client');
 go('clientDashboard');
}

function acceptWork(){
 alert('Delivery accepted. Funds released.');
 go('payout');
}
